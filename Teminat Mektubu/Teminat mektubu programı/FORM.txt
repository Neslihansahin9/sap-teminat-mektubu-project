*&      Module  USER_COMMAND_0100  INPUT
*&---------------------------------------------------------------------*
MODULE user_command_0100 INPUT.
  CLEAR gv_ekbelge.

  CASE sy-ucomm.
    WHEN 'SAVE'.
      PERFORM f_check_fields.
      IF gv_error IS INITIAL.
        PERFORM f_popup USING TEXT-q01.
        PERFORM f_save.
      ENDIF.

      IF zfi_teminat-belnr IS NOT INITIAL.
        gv_lock_fields = abap_true.
      ENDIF.

      gv_control_active = abap_false.

    WHEN 'OTHER'.
      PERFORM f_other.

    WHEN 'NEW'.
      PERFORM f_new.

    WHEN 'IADE'.
      PERFORM f_iade.

    WHEN 'SHOW'.
      PERFORM f_show.

    WHEN 'CONTROL'.
      PERFORM f_control.

    WHEN 'KOMISYON'.
      PERFORM f_call_screen_400.

    WHEN 'LOG'.
      PERFORM f_popup_alv.

    WHEN 'B1'.
      PERFORM f_show_doc USING zfi_teminat-bukrs
                               zfi_teminat-belnr
                               zfi_teminat-gjahr.
    WHEN 'B2'.
      PERFORM f_show_doc USING zfi_teminat-bukrs
                               zfi_teminat-stblg
                               zfi_teminat-stjah.
    WHEN 'B3'.
      PERFORM f_file_upload.

    WHEN 'B4'.
      PERFORM f_view_file.

    WHEN 'B5'.
      PERFORM f_show_doc USING zfi_teminat-bukrs
                               zfi_teminat-belnr2
                               zfi_teminat-gjahr.
    WHEN 'B6'.
      PERFORM f_show_doc USING zfi_teminat-bukrs
                               zfi_teminat-stblg
                               zfi_teminat-gjahr.
    WHEN 'EKBELGE'.
      gv_ekbelge = 'X'.
      PERFORM f_check_fields.
      IF gv_error IS INITIAL.
        PERFORM f_popup USING TEXT-q01.
        PERFORM f_save.
      ENDIF.

*    WHEN 'KISMICIKIS'.
*      PERFORM f_kismicikis.

    WHEN 'BACK'.
      SET SCREEN 0.
      LEAVE SCREEN.

    WHEN 'CANCEL'.
      LEAVE PROGRAM.

    WHEN 'EXIT'.
      LEAVE PROGRAM.

    WHEN 'YAZDIR'.
      PERFORM yazdir.

    WHEN 'TRSYAZDIR'.
      PERFORM iade_yazdir.
  ENDCASE.

ENDMODULE.                 " USER_COMMAND_0100  INPUT

*&      Form  F_DOCNO_POPUP
*&---------------------------------------------------------------------*
FORM f_docno_popup.

  TYPES : BEGIN OF ty_docno,
            docno TYPE zfi_teminat-docno,
            bukrs TYPE zfi_teminat-bukrs,
            name1 TYPE zfi_teminat-name1,
            aktut TYPE zfi_teminat-aktut,
          END OF ty_docno.

  DATA : lt_docnos TYPE TABLE OF zfi_teminat,
         lt_docno  TYPE TABLE OF ty_docno,
         ls_docno  TYPE ty_docno,
         lt_return TYPE TABLE OF ddshretval,
         ls_return TYPE ddshretval,
         lt_fields TYPE TABLE OF sval,
         ls_field  TYPE sval.

  FIELD-SYMBOLS <ls_docnos> TYPE zfi_teminat.

  IF r_at = 'X'.
    SELECT *
      FROM zfi_teminat
      INTO TABLE @lt_docnos
      WHERE kunnr IS NOT NULL AND kunnr <> ''.
  ELSEIF r_vt = 'X'.
    SELECT *
      FROM zfi_teminat
      INTO TABLE @lt_docnos
      WHERE lifnr IS NOT NULL AND lifnr <> ''.
  ENDIF.

  SORT lt_docnos BY lifnr kunnr.

  CLEAR lt_docno.
  LOOP AT lt_docnos ASSIGNING <ls_docnos>.
    CLEAR ls_docno.
    ls_docno-docno = <ls_docnos>-docno.
    ls_docno-bukrs = <ls_docnos>-bukrs.
    ls_docno-name1 = <ls_docnos>-name1.
    ls_docno-aktut = <ls_docnos>-aktut.
    APPEND ls_docno TO lt_docno.
  ENDLOOP.

  CLEAR lt_fields. REFRESH lt_fields.

  ls_field-tabname   = 'ZFI_TEMINAT'.
  ls_field-fieldname = 'DOCNO'.
  APPEND ls_field TO lt_fields.

  ls_field-tabname   = 'ZFI_TEMINAT'.
  ls_field-fieldname = 'BUKRS'.
  APPEND ls_field TO lt_fields.

  ls_field-tabname   = 'ZFI_TEMINAT'.
  ls_field-fieldname = 'NAME1'.
  APPEND ls_field TO lt_fields.

  ls_field-tabname   = 'ZFI_TEMINAT'.
  ls_field-fieldname = 'AKTUT'.
  APPEND ls_field TO lt_fields.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield        = 'DOCNO'
      dynpprog        = sy-repid
      dynpnr          = sy-dynnr
      value_org       = 'S'
    TABLES
      value_tab       = lt_docno
      return_tab      = lt_return
    EXCEPTIONS
      parameter_error = 1
      no_values_found = 2
      OTHERS          = 3.

  IF sy-subrc = 0 AND lt_return IS NOT INITIAL.
    READ TABLE lt_return INTO ls_return INDEX 1.
    IF sy-subrc = 0.
      p_docno  = ls_return-fieldval.
      gv_docno = p_docno.
    ENDIF.
  ENDIF.

ENDFORM.                    " F_DOCNO_POPUP

*&      Form  F_GET_DATA
*&---------------------------------------------------------------------*
FORM f_get_data USING p_mode.

  DATA lv_isltr LIKE zfi_teminat-isltr.

  IF gv_docno IS INITIAL.
    MESSAGE |Döküman No Giriniz| TYPE 'S' DISPLAY LIKE 'E'.
  ELSE.
    CLEAR zfi_teminat.
    PERFORM f_isltr CHANGING lv_isltr.

    SELECT SINGLE *
      FROM zfi_teminat
      WHERE docno = gv_docno AND
            isltr = lv_isltr.

    IF sy-subrc = 0.
      gv_mode = p_mode.
    ELSE.
      zfi_teminat-docno = '$000000001'.
      MESSAGE |Belge Türünü kontrol ediniz.| TYPE 'S' DISPLAY LIKE 'E'.
    ENDIF.
  ENDIF.

ENDFORM.                    " F_GET_DATA

      number_range_not_intern = 2
      object_not_found        = 3
      quantity_is_0           = 4
      quantity_is_not_1       = 5
      interval_overflow       = 6

    zfi_teminat-docno = p_docno.
  ELSE.
    MESSAGE |Döküman No Türetilemedi!| TYPE 'S' DISPLAY LIKE 'E'.
  ENDIF.

*&---------------------------------------------------------------------*
*&      Form  F_USER_COMMAND
*&---------------------------------------------------------------------*
FORM f_user_command USING r_ucomm TYPE sy-ucomm
                      rs_selfield TYPE slis_selfield.

  DATA lv_error TYPE abap_bool VALUE abap_false.

  CASE r_ucomm.
    WHEN 'KAYIT'.
      LOOP AT gt_log ASSIGNING <fs_data> WHERE belnr IS INITIAL.
        PERFORM f_teminat_bapi_log.
        PERFORM f_save_db_teminat.
      ENDLOOP.

      PERFORM f_show_message_teminat.

      COMMIT WORK.
      MESSAGE |Kayıt işlemi başarıyla tamamlandı.| TYPE 'S'.
      rs_selfield-refresh = 'X'.

    WHEN 'TERSKAYIT'.
      LOOP AT gt_log ASSIGNING <fs_data>.
        PERFORM f_reverse_document_log.
        IF sy-subrc <> 0.
          lv_error = abap_true.
        ENDIF.
        PERFORM f_save_db_teminat.
        IF sy-subrc <> 0.
          lv_error = abap_true.
        ENDIF.
      ENDLOOP.

      IF lv_error = abap_false.
        COMMIT WORK.
        MESSAGE |Ters Kayıt işlemi başarıyla tamamlandı.| TYPE 'S'.
      ELSE.
        MESSAGE |Ters Kayıt işlemi tamamlanamadı!| TYPE 'E'.
      ENDIF.
  ENDCASE.

ENDFORM.                 " F_USER_COMMAND

*&      Form  F_FILL_RECEIVABLE_TEMINAT
*&---------------------------------------------------------------------*
FORM f_fill_receivable_teminat.

  IF <fs_data>-lifnr IS NOT INITIAL.
    DATA(lv_koart) = 'K'.
  ELSEIF <fs_data>-kunnr IS NOT INITIAL.
    lv_koart = 'D'.
  ENDIF.

  SELECT SINGLE *
    FROM zfi_t_tem_odk_tr
    INTO @DATA(ls_odk)
    WHERE bukrs = @<fs_data>-bukrs AND
          koart = @lv_koart        AND
          umskz = @<fs_data>-umskz_odk.

  ADD 1 TO itemno_acc.
  accountreceivable-itemno_acc = itemno_acc.
  accountreceivable-customer   = <fs_data>-kunnr.
  accountreceivable-bline_date = <fs_data>-akotr.
  accountreceivable-item_text  = <fs_data>-konu.
  accountreceivable-sp_gl_ind  = <fs_data>-umskz_odk.
  accountreceivable-comp_code  = <fs_data>-bukrs.
  APPEND accountreceivable. CLEAR accountreceivable.

  currencyamount-itemno_acc   = itemno_acc.
  currencyamount-currency_iso = <fs_data>-waers.

** Ters mantık...
  IF ls_odk-shkzg EQ 'S'.
    currencyamount-amt_doccur = <fs_data>-aktut * -1.
  ELSEIF ls_odk-shkzg EQ 'H'.
    currencyamount-amt_doccur = <fs_data>-aktut.
  ENDIF.
  APPEND currencyamount. CLEAR currencyamount.

ENDFORM.                    " F_FILL_RECEIVABLE_TEMINAT

*&      Form  F_CALL_BAPI_TEMINAT
*&---------------------------------------------------------------------*
FORM f_call_bapi_teminat.

  IF <fs_data>-lifnr IS NOT INITIAL.
    CALL FUNCTION 'BAPI_ACC_DOCUMENT_POST'
      EXPORTING
        documentheader = header
      IMPORTING
        obj_type       = obj_type
        obj_key        = obj_key
        obj_sys        = obj_sys
      TABLES
        accountgl      = accountgl
        accountpayable = accountpayable
        currencyamount = currencyamount
        return         = return.

  ELSEIF <fs_data>-kunnr IS NOT INITIAL.
    CALL FUNCTION 'BAPI_ACC_DOCUMENT_POST'
      EXPORTING
        documentheader    = header
      IMPORTING
        obj_type          = obj_type
        obj_key           = obj_key
        obj_sys           = obj_sys
      TABLES
        accountgl         = accountgl
        accountreceivable = accountreceivable
        currencyamount    = currencyamount
        return            = return.
  ENDIF.

  APPEND LINES OF return TO gt_mess_all.

  LOOP AT return WHERE type CA 'EAX'.
    gv_error = 'X'.
    EXIT.
  ENDLOOP.

  IF obj_key NE space AND obj_key NE '$'.
    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
      EXPORTING
        wait = 'X'.

    <fs_data>-belnr = obj_key(10).
    <fs_data>-gjahr = obj_key+14(4).
  ELSE.
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
    gv_error = 'X'.
  ENDIF.

ENDFORM.                    " F_CALL_BAPI_TEMINAT
*&---------------------------------------------------------------------*
*&      Form  F_SHOW_MESSAGE_TEMINAT
*&---------------------------------------------------------------------*
FORM f_show_message_teminat.

  IF gv_error = 'X' OR gt_mess_all[] IS NOT INITIAL.
    LOOP AT gt_mess_all.
      CLEAR wa_mess.
      wa_mess-msgid = gt_mess_all-id.
      wa_mess-msgty = gt_mess_all-type.
      wa_mess-msgno = gt_mess_all-number.
      wa_mess-msgv1 = gt_mess_all-message_v1.
      wa_mess-msgv2 = gt_mess_all-message_v2.
      wa_mess-msgv3 = gt_mess_all-message_v3.
      wa_mess-msgv4 = gt_mess_all-message_v4.
      APPEND wa_mess TO gt_mess.
    ENDLOOP.

    IF gt_mess[] IS NOT INITIAL.
      CALL FUNCTION 'C14Z_MESSAGES_SHOW_AS_POPUP'
        TABLES

*&      Module  MODIFY_SCREEN  OUTPUT
*&---------------------------------------------------------------------*
MODULE modify_screen OUTPUT.

  IF zfi_teminat-kunnr IS NOT INITIAL.
    SELECT SINGLE name1
      FROM kna1
      INTO zfi_teminat-name1
      WHERE kunnr = zfi_teminat-kunnr.

    zfi_teminat-koart = 'D'.
  ENDIF.

  IF zfi_teminat-lifnr IS NOT INITIAL.
    SELECT SINGLE name1
      FROM lfa1
      INTO zfi_teminat-name1
      WHERE lifnr = zfi_teminat-lifnr.

    zfi_teminat-koart = 'K'.
  ENDIF.

  IF zfi_teminat-kunnr IS INITIAL
    AND zfi_teminat-lifnr IS INITIAL.
    zfi_teminat-koart = ''.
  ENDIF.

  IF zfi_teminat-bukrs IS INITIAL.
    CLEAR zfi_teminat-butxt.
  ELSE.
    SELECT SINGLE butxt
      FROM t001
      INTO zfi_teminat-butxt
      WHERE bukrs = zfi_teminat-bukrs.
  ENDIF.

  CASE gv_mode.
    WHEN 'N' OR 'C'.
      LOOP AT SCREEN.
        IF screen-name = 'ZFI_TEMINAT-DOCNO' OR
           screen-name = 'ZFI_TEMINAT-NAME1' OR
           screen-name = 'ZFI_TEMINAT-DMBTR' OR
           screen-name = 'ZFI_TEMINAT-BLART'.
        ELSE.
          screen-input = 1.
        ENDIF.
        MODIFY SCREEN.
      ENDLOOP.
    WHEN 'S' OR 'D'.
      LOOP AT SCREEN.
        IF screen-name NE 'B1'     AND
           screen-name NE 'B2'     AND
           screen-name NE 'B5'     AND
           screen-name NE 'B6'     AND
           screen-name NE 'YAZDIR' AND
           screen-name NE 'TRSYAZDIR'.
          screen-input = 0.
          MODIFY SCREEN.
        ENDIF.
      ENDLOOP.
  ENDCASE.

  IF r_at = 'X' OR r_vt = 'X'.
    LOOP AT SCREEN.
      IF screen-name = 'ZFI_TEMINAT-AKYTR'.
        screen-active = 0.
        MODIFY SCREEN.
      ENDIF.
    ENDLOOP.
  ENDIF.

  LOOP AT SCREEN.
    IF gv_mode = 'S' OR gv_mode = 'D'.
      IF screen-name = 'ZFI_TEMINAT-BIRIM_TEXT'
      OR screen-name = 'ZFI_TEMINAT-URUN_TEXT'.
        screen-input = 0.
        MODIFY SCREEN.
      ENDIF.
    ELSE.
      IF screen-name = 'ZFI_TEMINAT-BIRIM_TEXT'.
        IF zfi_teminat-tebirim = 'DIĞER'.
          screen-input = 1.
        ELSE.
          screen-input = 0.
          CLEAR zfi_teminat-birim_text.
        ENDIF.
        MODIFY SCREEN.
      ENDIF.

      IF screen-name = 'ZFI_TEMINAT-URUN_TEXT'.
        IF zfi_teminat-teurun = 'DIĞER'.
          screen-input = 1.
        ELSE.
          screen-input = 0.
          CLEAR zfi_teminat-urun_text.
        ENDIF.
        MODIFY SCREEN.
      ENDIF.
    ENDIF.
  ENDLOOP.

  LOOP AT SCREEN.
    IF screen-name = 'ZFI_TEMINAT-BELNR'
      AND zfi_teminat-belnr IS NOT INITIAL.
      screen-input = 0.
      MODIFY SCREEN.
    ENDIF.
    IF screen-name = 'ZFI_TEMINAT-BELNR2'
      AND zfi_teminat-belnr2 IS NOT INITIAL.
      screen-input = 0.
      MODIFY SCREEN.
    ENDIF.
    IF screen-name = 'ZFI_TEMINAT-STBLG'
      AND zfi_teminat-stblg IS NOT INITIAL.
      screen-input = 0.
      MODIFY SCREEN.
    ENDIF.
  ENDLOOP.

  LOOP AT SCREEN.
    IF screen-name = 'ZFI_TEMINAT-TOTALMASRAF'
    OR screen-name = 'ZFI_TEMINAT-BSMW'.
      screen-input = 0.
      MODIFY SCREEN.
    ENDIF.
  ENDLOOP.

  LOOP AT SCREEN.
    IF screen-name = 'ZFI_TEMINAT-KMSYTUTARI'.
      IF gv_control_active = abap_true.
        screen-input = 1.
      ELSE.
        screen-input = 0.
      ENDIF.
      MODIFY SCREEN.
    ENDIF.
  ENDLOOP.

  LOOP AT SCREEN.
    IF r_at = 'X' AND
         ( screen-name = 'ZFI_TEMINAT-LIFNR'
        OR screen-name = 'ZFI_TEMINAT-BELNR2'
        OR screen-name = 'ZFI_TEMINAT-KMSYORANI'
        OR screen-name = 'ZFI_TEMINAT-KMSYTUTARI'
        OR screen-name = 'ZFI_TEMINAT-BSMW'
        OR screen-name = 'ZFI_TEMINAT-TOTALMASRAF'
        OR screen-name = 'ZFI_TEMINAT-ANAHESAP' ).
      screen-input = 0.
      MODIFY SCREEN.
    ELSEIF r_vt = 'X' AND screen-name = 'ZFI_TEMINAT-KUNNR'.
      screen-input = 0.
      MODIFY SCREEN.
    ENDIF.
  ENDLOOP.

  zfi_teminat-blart = 'TM'.
  zfi_teminat-bsmw  = '0,05'.

  IF gv_lock_fields = abap_true.
    LOOP AT SCREEN.
      IF ( screen-name = 'ZFI_TEMINAT-BANKA'       )
      OR ( screen-name = 'ZFI_TEMINAT-BUKRS'       )
      OR ( screen-name = 'ZFI_TEMINAT-BUTXT'       )
      OR ( screen-name = 'ZFI_TEMINAT-KRRFN'       )
      OR ( screen-name = 'ZFI_TEMINAT-AKTUT'       )
      OR ( screen-name = 'ZFI_TEMINAT-WAERS'       )
      OR ( screen-name = 'ZFI_TEMINAT-AKATR'       )
      OR ( screen-name = 'ZFI_TEMINAT-AKOTR'       )
      OR ( screen-name = 'ZFI_TEMINAT-UMSKZ_ODK'   )
      OR ( screen-name = 'ZFI_TEMINAT-KONU'        )
      OR ( screen-name = 'ZFI_TEMINAT-TSURESI'     )
      OR ( screen-name = 'ZFI_TEMINAT-KMSYORANI'   )
      OR ( screen-name = 'ZFI_TEMINAT-MKMSYORANI'  )
      OR ( screen-name = 'ZFI_TEMINAT-TEBIRIM'     )
      OR ( screen-name = 'ZFI_TEMINAT-TEKISI'      )
      OR ( screen-name = 'ZFI_TEMINAT-TEURUN'      )
      OR ( screen-name = 'ZFI_TEMINAT-TMTUR'       )
      OR ( screen-name = 'ZFI_TEMINAT-TMTUR2'      )
      OR ( screen-name = 'ZFI_TEMINAT-BRNCH'       )
      OR ( screen-name = 'ZFI_TEMINAT-KUNNR'       )
      OR ( screen-name = 'ZFI_TEMINAT-LIFNR'       )
      OR ( screen-name = 'ZFI_TEMINAT-ANAHESAP'    )
      OR ( screen-name = 'ZFI_TEMINAT-BIRIM_TEXT'  )
      OR ( screen-name = 'ZFI_TEMINAT-URUN_TEXT'   ).
        screen-input = 0.
        MODIFY SCREEN.
      ENDIF.
    ENDLOOP.
  ENDIF.

*&      Form  F_IADE
*&---------------------------------------------------------------------*
FORM f_iade.

  IF zfi_teminat-belnr IS INITIAL.
    MESSAGE |Lütfen Belge Seçiniz!| TYPE 'S' DISPLAY LIKE 'E'.
  ELSE.
    PERFORM f_iade_tarihi.

    IF gv_iade_cancel = abap_true.
      MESSAGE |İşlem kullanıcı tarafından iptal edildi!| TYPE 'S' DISPLAY LIKE 'E'.
      RETURN.
    ENDIF.

    IF gv_iade_tarihi IS INITIAL.
      MESSAGE |İade Tarihi Giriniz!| TYPE 'S' DISPLAY LIKE 'E'.
      RETURN.
    ELSE.
      PERFORM f_popup USING TEXT-q02.

      IF zfi_teminat-stblg IS INITIAL.
        PERFORM f_reverse_document.
      ELSE.
        MESSAGE |Belgenin ters kaydı alınmıştır.| TYPE 'E'.
      ENDIF.
    ENDIF.
  ENDIF.

  RETURN.

*&---------------------------------------------------------------------*
*&      Form  F_IADE_TARIHI
*&---------------------------------------------------------------------*
FORM f_iade_tarihi.

  DATA : lv_returncode      TYPE char4,
         lt_fields          TYPE TABLE OF sval WITH HEADER LINE,
         lv_return_date     TYPE sy-datum,
         lv_prev_month_end  TYPE sy-datum,
         lv_current_day_num TYPE i.

  CLEAR gv_iade_tarihi.

  lt_fields-tabname   = 'BKPF'.
  lt_fields-fieldname = 'BUDAT'.
  lt_fields-field_obl = 'X'.
  lt_fields-fieldtext = 'İade Tarihi'.
  APPEND lt_fields. CLEAR lt_fields.

  CALL FUNCTION 'POPUP_GET_VALUES'
    EXPORTING
      popup_title = 'İade Tarihi Giriniz'
    IMPORTING
      returncode  = lv_returncode
    TABLES
      fields      = lt_fields.

  IF lv_returncode = 'A'.
    gv_iade_cancel = abap_true.
    MESSAGE |İşlem İptal Edildi| TYPE 'S' DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

  IF sy-subrc = 0 AND lt_fields[] IS NOT INITIAL.
    READ TABLE lt_fields INDEX 1.

    lv_return_date = lt_fields-value.
    lv_current_day_num = sy-datum+6(2).

    DATA(lv_current_month_first_day) = sy-datum.
    lv_current_month_first_day+6(2) = '01'.

    CALL FUNCTION 'CALCULATE_DATE'
      EXPORTING
        days        = -1
        start_date  = lv_current_month_first_day
      IMPORTING
        result_date = lv_prev_month_end.

    IF lv_current_day_num > 15 AND lv_return_date <= lv_prev_month_end.
      MESSAGE |İade tarihi, önceki aya ait olamaz. Dönem kapanışı yapılmıştır.| TYPE 'E'.
      CLEAR gv_iade_tarihi.
      EXIT.
    ELSE.
      gv_iade_tarihi = lv_return_date.
    ENDIF.
  ENDIF.

ENDFORM.                    " F_IADE_TARIHI

FORM f_control.

         lv_kmsorani    TYPE p DECIMALS 6,
         lv_bsmw2       TYPE p LENGTH 16 DECIMALS 6,
         lv_bsmw        TYPE p LENGTH 16 DECIMALS 2,
         lv_bsmw_char   TYPE string.
*        ls_bsmw        TYPE zfi_t_bsmw.

* Komisyon Oranı
  CALL FUNCTION 'MOVE_CHAR_TO_NUM'
    EXPORTING
      chr             = zfi_teminat-kmsyorani
    IMPORTING
      num             = lv_kmsorani
    EXCEPTIONS
      convt_no_number = 1
      convt_overflow  = 2
      OTHERS          = 3.
  IF sy-subrc <> 0.
  ENDIF.

    lv_kmsytutari = zfi_teminat-aktut * lv_kmsorani.
  ELSE.
    lv_kmsytutari = zfi_teminat-kmsytutari.
  ENDIF.

* BSMW
  CALL FUNCTION 'MOVE_CHAR_TO_NUM'
    EXPORTING
      chr             = zfi_teminat-bsmw
    IMPORTING
      num             = lv_bsmw2
    EXCEPTIONS
      convt_no_number = 1
      convt_overflow  = 2
      OTHERS          = 3.
  IF sy-subrc <> 0.
  ENDIF.

*  SELECT SINGLE *
*    FROM zfi_t_bsmw
*    INTO ls_bsmw
*    WHERE bsmw = zfi_teminat-bsmw.
*
*  IF sy-subrc <> 0.
*    CLEAR zfi_teminat-bsmw.
*    MESSAGE |Girilen BSMW değeri, BSMW bakım tablosunda bulunmamaktadır!| TYPE 'E'.
*  ENDIF.

  lv_bsmw = lv_kmsytutari * lv_bsmw2.

  zfi_teminat-kmsytutari  = lv_kmsytutari.

  lv_bsmw_char = lv_bsmw.
  CONDENSE lv_bsmw_char NO-GAPS.
  zfi_teminat-bsmw = lv_bsmw_char.

  lv_totalmasraf = zfi_teminat-kmsytutari + lv_bsmw.

  IF zfi_teminat-tsuresi = '1 YILLIK'.
    lv_totalmasraf = lv_totalmasraf / 1.
  ELSEIF zfi_teminat-tsuresi = '6 AYLIK'.
    lv_totalmasraf = lv_totalmasraf / 2.
  ELSEIF zfi_teminat-tsuresi = '3 AYLIK'.
    lv_totalmasraf = lv_totalmasraf / 4.
  ENDIF.

  zfi_teminat-totalmasraf = lv_totalmasraf.

  gv_control_active = abap_true.
  LEAVE TO SCREEN sy-dynnr.

*&---------------------------------------------------------------------*
FORM f_komisyon_hesabi.

  DATA : lv_date   TYPE sy-datum,
         lv_akotr  TYPE sy-datum,
         lv_akatr  TYPE sy-datum,
         lv_months LIKE vtbbewe-atage,
         lv_div    TYPE p DECIMALS 2.

  CLEAR itab. REFRESH itab.

  CALL SELECTION-SCREEN 201 STARTING AT 10 10.

  CHECK sy-subrc = 0.
  CONCATENATE p_gjahr '0101' INTO lv_akotr.
  CONCATENATE p_gjahr '1231' INTO lv_akatr.

  SELECT *
    FROM zfi_teminat
    INTO CORRESPONDING FIELDS OF TABLE itab
    WHERE bukrs IN s_bukrs  AND
          isltr IN s_isltr  AND
          akatr <= lv_akatr AND
          akotr >= lv_akotr AND
          kmsik NE space.

  IF itab[] IS INITIAL.
    MESSAGE |Aradığınız kriterlerde kayıt bulunamadı!| TYPE 'I'.
  ELSE.
    PERFORM f_get_month.
    LOOP AT itab ASSIGNING <fs_itab>.
      lv_date =  <fs_itab>-akatr.
*yıl dönmüşse bi sonraki yılın ocak ayından başlaması için...
      IF <fs_itab>-kmsik = '5'. "yıllık
        lv_date(4) = lv_date(4) + 1.
        IF lv_date(4) NE p_gjahr.
          CONTINUE.
        ENDIF.
      ELSE.
        IF lv_date(4) < p_gjahr AND <fs_itab>-kmsik NE 1.
          CONCATENATE p_gjahr '01' lv_date+6(2) INTO lv_date.
        ENDIF.
      ENDIF.
      CASE <fs_itab>-kmsik.
        WHEN '1'. "peşin
          PERFORM f_assign_field USING lv_date '1'.
        WHEN '2'. "her ay
          PERFORM f_find_months USING lv_date <fs_itab>-akotr
                             CHANGING lv_months.
          PERFORM f_assign_field USING lv_date '12'.
          DO lv_months TIMES.
            PERFORM f_add_month  USING lv_date '1'
                              CHANGING lv_date.
            PERFORM f_assign_field USING lv_date '12'.
          ENDDO.
        WHEN '3'. "3 ayda 1
          PERFORM f_find_months USING lv_date <fs_itab>-akotr
                             CHANGING lv_months.
          lv_div = lv_months / 3.
          lv_months = trunc( lv_div ).
          PERFORM f_assign_field USING lv_date '4'.
          DO lv_months TIMES.
            PERFORM f_add_month  USING lv_date '3'
                              CHANGING lv_date.
            PERFORM f_assign_field USING lv_date '4'.
          ENDDO.
        WHEN '4'. "2 ayda 1
          PERFORM f_find_months USING lv_date <fs_itab>-akotr
                             CHANGING lv_months.
          lv_div = lv_months / 2.
          lv_months = trunc( lv_div ).
          PERFORM f_assign_field USING lv_date '6'.
          DO lv_months TIMES.
            PERFORM f_add_month  USING lv_date '2'
                              CHANGING lv_date.
            PERFORM f_assign_field USING lv_date '6'.
          ENDDO.
        WHEN '5'. "yıllık
          PERFORM f_assign_field USING lv_date '1'.
      ENDCASE.
    ENDLOOP.
    CALL SCREEN 200.
  ENDIF.

ENDFORM.                    " F_KOMISYON_HESABI

FORM f_assign_field USING p_date p_divide.

  ELSE.

  ENDIF.

         END OF lt_msj.

  CLEAR : gt_bdcdata[], lt_msj[].

  LOOP AT lt_teminat INTO ls_teminat.

      p  'SAPMF05L'                       '0102'.

      f : 'BDC_CURSOR'                   'RF05L-BUZEI',
          'BDC_OKCODE'                   '/00',
          'RF05L-BELNR'                  ls_teminat-belnr,
          'RF05L-BUKRS'                  ls_teminat-bukrs,

          'RF05L-BUZEI'                  ' ',
          'RF05L-XKKRE'                  'X'.

      p   'SAPMF05L'                     '0303'.
      f : 'BDC_CURSOR'                   'BSEG-ZFBDT',
          'BDC_OKCODE'                   '=AE',
          'BSEG-ZFBDT'                   lv_vade_out.
    ELSEIF ls_teminat-koart = 'D'.
      p  'SAPMF05L'                       '0102'.

      f : 'BDC_CURSOR'                   'RF05L-BUZEI',

          'BDC_OKCODE'                   '=AE',
          'BSEG-ZFBDT'                   lv_vade_out.
    ENDIF.

    CALL TRANSACTION 'FB09' USING gt_bdcdata
    MESSAGES INTO lt_messtab MODE lv_mode.

    READ TABLE lt_messtab INTO ls_messtab INDEX 1.
    IF sy-subrc = 0.
      lt_msj-belnr = ls_teminat-belnr.
      lt_msj-type  = ls_messtab-msgtyp.
      APPEND lt_msj.
      IF ls_messtab-msgtyp CN 'AXE'.

        CALL FUNCTION 'CONVERT_DATE_TO_INTERNAL'
          EXPORTING
            date_external = lv_vade_out
          IMPORTING
            date_internal = l_date.

        UPDATE zfi_teminat
          SET akotr = l_date
        WHERE gjahr = ls_teminat-gjahr
          AND bukrs = ls_teminat-bukrs
          AND krrfn = ls_teminat-krrfn
          AND docno = ls_teminat-docno.
      ENDIF.
    ENDIF.

    CLEAR : gt_bdcdata[], gt_bdcdata, lt_msj, ls_messtab, lt_messtab[],
            ls_teminat.
  ENDLOOP.

  LOOP AT lt_msj WHERE type = 'E'.
    CONCATENATE lt_msj-belnr '-' INTO lv_belnr.
  ENDLOOP.

  IF sy-subrc = 0.
    lv_belnr = lv_belnr && |Numaralı Belgeler Değiştirilemedi! Belgeleri Kontrol Ediniz!|.
    MESSAGE lv_belnr TYPE 'S' DISPLAY LIKE 'E'.
    MESSAGE s031(zfi).
  ENDIF.
