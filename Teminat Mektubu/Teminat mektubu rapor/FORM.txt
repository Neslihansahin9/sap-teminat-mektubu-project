*& Form F_PREPARE_FLDCT
*&---------------------------------------------------------------------*
FORM f_prepare_fldct USING u_sname TYPE tabname.
  DATA ls_fldct LIKE lvc_s_fcat.

  FREE gt_fldct.
  CLEAR gs_layout_0100.

  gs_layout_0100-zebra      = 'X'.
  gs_layout_0100-cwidth_opt = 'X'.
  gs_layout_0100-sel_mode   = 'A'.
  gs_layout_0100-no_rowins  = 'X'.

  gs_variant_0100-report = sy-repid.
  gs_variant_0100-handle = '1'.

  CALL FUNCTION 'LVC_FIELDCATALOG_MERGE'
    EXPORTING
      i_structure_name       = u_sname
      i_bypassing_buffer     = 'X'
    CHANGING
      ct_fieldcat            = gt_fldct
    EXCEPTIONS
      inconsistent_interface = 1
      program_error          = 2
      OTHERS                 = 3.

  DEFINE edit.
    ls_fldct-edit = 'X'.
  MODIFY gt_fldct FROM ls_fldct TRANSPORTING edit WHERE fieldname EQ &1.
  END-OF-DEFINITION.

  LOOP AT gt_fldct REFERENCE INTO DATA(lrd_fldct).
    CASE lrd_fldct->fieldname.
      WHEN 'DMBTR' OR 'WRBTR_TXT' OR 'WRBTR'.
        lrd_fldct->no_out = 'X'.
      WHEN 'BELNR'.
        IF p_iade EQ 'X'.
          lrd_fldct->no_out = 'X'.
        ENDIF.
      WHEN 'BELNR2'.
        IF p_iade EQ 'X'.
          lrd_fldct->no_out = 'X'.
        ENDIF.
      WHEN 'KUNNR'.
        IF p_rdl EQ 'X'.
          lrd_fldct->no_out = 'X'.
        ENDIF.
      WHEN 'LIFNR'.
        IF p_rdk EQ 'X'.
          lrd_fldct->no_out = 'X'.
        ENDIF.
      WHEN 'TMTUR'.
        lrd_fldct->seltext = 'Tür'.
        lrd_fldct->coltext = 'Tür'.
        lrd_fldct->reptext = 'Tür'.
      WHEN 'TMTUR2'.
        lrd_fldct->seltext = 'Tür 2'.
        lrd_fldct->coltext = 'Tür 2'.
        lrd_fldct->reptext = 'Tür 2'.
      WHEN 'STBLG' OR 'IADE_TARIHI'.
        IF p_tem EQ 'X'.
          lrd_fldct->no_out = 'X'.
        ENDIF.
      WHEN 'BLART2'.
        IF p_tem EQ 'X'.
          lrd_fldct->no_out = 'X'.
        ENDIF.
      WHEN 'BLART'.
        IF p_iade EQ 'X'.
          lrd_fldct->no_out = 'X'.
        ENDIF.
      WHEN 'HKONT1'.
        lrd_fldct->no_out = 'X'.
      WHEN 'AKTUT_TXT'.
        lrd_fldct->no_out = 'X'.
      WHEN 'URUN_TEXT'.
        IF zfi_teminat-urun_text IS INITIAL.
          lrd_fldct->no_out = 'X'.
        ENDIF.
      WHEN 'BIRIM_TEXT'.
        IF zfi_teminat-birim_text IS INITIAL.
          lrd_fldct->no_out = 'X'.
        ENDIF.
      WHEN 'VKNMUSTERI'.
        IF zfi_teminat-vknmusteri IS INITIAL.
          lrd_fldct->no_out = 'X'.
        ENDIF.
      WHEN 'VKNSATICI'.
        IF zfi_teminat-vknsatici IS INITIAL.
          lrd_fldct->no_out = 'X'.
        ENDIF.
      WHEN OTHERS.
    ENDCASE.
  ENDLOOP.

        screen-active  = 1.
      ENDIF.
      MODIFY SCREEN.
    ENDLOOP.
  ENDIF.

  LOOP AT SCREEN.
    IF screen-name = 'P_BLART'.
      IF p_tem = 'X'.
        p_blart = 'TM'.
      ELSEIF p_iade = 'X'.
        p_blart = 'TK'.
      ENDIF.
    ENDIF.
    MODIFY SCREEN.
  ENDLOOP.

ENDFORM.                    " SELECTION_SCREEN_OUTPUT
